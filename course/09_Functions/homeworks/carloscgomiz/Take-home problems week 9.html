ExtractKeywords <- function(x){
    Sample <- keyword(x)$`$FIL`
    Cytometer <- keyword(x)$`$CYT`
    CytometerSN <- keyword(x)$`$CYTSN`
    Operator <- keyword(x)$`$OP`
    BeginTime <- keyword(x)$`$BTIM`
    EndTime <- keyword(x)$`$ETIM`
    Volume <- keyword(x)$`$VOL`
    FlowRate <- keyword(x)$`$FLOWRATE`

    dataframe <- data.frame(
        Sample=Sample,
        Cytometer=Cytometer,
        CytometerSN =CytometerSN,
        Operator=Operator,
        BeginTime=BeginTime,
        EndTime=EndTime,
        Volume=Volume,
        FlowRate=FlowRate
    )
    return(dataframe)
}

For CellConcentration, we retrieved both start and end time. Look up information on lubridate package, convert these times to a time-style format, and from acquired volume derrive uL/min at which each .fcs file was acquired. Did this vary at all across days?

In this case, I will import the GatingSet and a summarized version of the function from the course:

#course.gs <- load_gs(file.path("Cytometry Carlos", "09_Functions", "data", "TheGatingSet.gs"))
course.gs <- load_gs("TheGatingSet.gs")

SummarizedFunction <- function(x){
    Date <- keyword(x)$`$DATE` # Day-Month-Year
    InstrumentSerial <- keyword(x)$`$CYTSN`
    TotalFileEvents <- keyword(x)$`$TOT`
    Specimen <- keyword(x)$GUID
    StartTime <- keyword(x)$`$BTIM`
    EndTime <- keyword(x)$`$ETIM`

    Volume <- keyword(x)$`$VOL`
    Volume <- as.numeric(Volume)

    Data <- data.frame(
        Date=Date,
        InstrumentSerial=InstrumentSerial,
        Specimen=Specimen,
        StartTime=StartTime,
        EndTime=EndTime,
        Volume=Volume
        )

    return(Data)
}
